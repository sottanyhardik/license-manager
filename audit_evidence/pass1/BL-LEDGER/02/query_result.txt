  from django.db import transaction
  from apps.trade.models import LicenseTradeLine
  from apps.bill_of_entry.models import RowDetails
  from apps.license.models import LicenseDetailsModel, LicenseBalance
  from apps.reconciliation.models import InvoiceBOEAllocation
  from apps.license.services.balance_calculator import LicenseBalanceCalculator as C

  lic = LicenseDetailsModel.objects.get(license_number='0311054433')
  line = LicenseTradeLine.objects.filter(trade_id=518).first()
  row = RowDetails.objects.get(id=274889)
  print('trade line', line.id, line.cif_fc, 'row', row.id, row.cif_fc)

  try:
      with transaction.atomic():
          before_stored = LicenseBalance.objects.get(license=lic).balance_cif
          before_debit = C.calculate_debit(lic)
          before_live_fin = C.calculate_financial_balance(lic)
          before_live_cust = C.calculate_balance(lic)
          print('BEFORE: stored', before_stored, 'calculate_debit', before_debit,
                'financial_balance', before_live_fin, 'customs_balance', before_live_cust)

          alloc = InvoiceBOEAllocation.objects.create(
              trade_line=line, row_details=row,
              allocated_qty=0, allocated_cif_fc=10000,
              status=InvoiceBOEAllocation.STATUS_ACTIVE, is_current=True,
          )

          lic.refresh_from_db()
          after_stored = LicenseBalance.objects.get(license=lic).balance_cif
          after_debit = C.calculate_debit(lic)
          after_live_fin = C.calculate_financial_balance(lic)
          after_live_cust = C.calculate_balance(lic)
          print('AFTER create alloc: stored', after_stored, 'calculate_debit', after_debit,
                'financial_balance', after_live_fin, 'customs_balance', after_live_cust)
          raise Exception('rollback - test only, no real change made')
  except Exception as e:
      print('rolled back:', e)
  "
